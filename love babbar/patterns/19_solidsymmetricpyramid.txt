void printPattern(int n) {
        for(int i=1; i<=n; i++){
            //part 1s number triangle
            for(int j=1;j<=i; j++){
                cout<<j;
            }
            //space
            for(int j=1; j<=2*(n-i); j++){
                cout<<" ";
            }
            //numbers reverse
            for(int j=i; j>=1; j--){
                cout<<j;
            }
            cout<<endl;
        }
    }
