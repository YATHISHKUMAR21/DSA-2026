class Solution {
public:
    void printReverseAlphabetTriangle(int n) {
        // Implement the pattern printing logic here
        for(int row=1; row<=n;row++){
            char ch='A';
            for(int col=1;col<=row;col++){
                cout<<char(ch+n-row+col-1);
            }
            cout<<endl;
        }
    }
};
