function selection_sort(int arr,int n){
    //select min index and swap with first ele of unsorted portion
    if(n<=1) return;
    for(int i=0; i<=n-2; i++){
        min=i;
        for(int j=i; j<=n-1; j++){
            if(arr[j]<arr[min]){
                min=j
            }
        }
        //after j loop comparision is done , min index will get updated
        so swap it with first ele which was arr[i]

        swap(arr[min],arr[i]);
        or
        temp = arr[i]
        arr[i]=arr[min]
        arr[min]=temp;
    }
}
